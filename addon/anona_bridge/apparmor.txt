# AppArmor profile for the Anona Pano Bridge add-on.
# The profile name must match the add-on slug (anona_bridge). Home Assistant
# loads this automatically when the file is present.
#include <tunables/global>

profile anona_bridge flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities the base-image init / housekeeping may use. (The bridge binds
  # ports > 1024 and drops nothing, so no NET_BIND_SERVICE etc. is required.)
  capability setuid,
  capability setgid,
  capability dac_override,
  capability chown,
  capability fowner,
  capability fsetid,
  capability kill,

  # Networking: TCP + UDP over IPv4/IPv6 for the RTSP/HTTP servers, the cloud
  # HTTPS/WSS calls, and WebRTC (STUN/TURN). netlink is needed so pion can
  # enumerate interfaces for ICE candidate gathering. Raw/packet sockets are
  # intentionally not allowed.
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  network netlink raw,

  # Signal the ffmpeg child used for talkback transcoding.
  signal (send) set=(kill,term,int,hup,cont),

  # Broad file access inside the container (shared libs, CA certs, DNS config,
  # add-on data). Execution is still restricted to the paths below.
  file,

  # s6-overlay (v3) init + bashio: the add-on's /run.sh runs under the base
  # image's s6 supervision tree because config.yaml sets init: false.
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/lib/bashio/** ix,
  /etc/s6-overlay/** ix,
  /package/** ix,
  /command/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,

  # The add-on itself.
  /run.sh r,
  /usr/bin/anona-bridge mrix,
  /usr/bin/ffmpeg mrix,

  # Runtime state and options (bashio reads /data/options.json).
  /data/** rw,
  /tmp/** rwk,
  /run/{,**} rwk,
  /dev/tty rw,
}
